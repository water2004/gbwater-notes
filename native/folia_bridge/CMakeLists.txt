set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b" CACHE STRING "" FORCE)
cmake_minimum_required(VERSION 4.0)
project(gbwater_folia_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FOLIA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/folia" CACHE PATH "Folia source tree")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(FOLIA_BUILD_MERMAID ON)
endif()
add_subdirectory("${FOLIA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/folia")
add_executable(folia_md bridge.cpp)
target_compile_features(folia_md PRIVATE cxx_std_23)
set_target_properties(folia_md PROPERTIES CXX_MODULE_STD ON)
target_link_libraries(folia_md PRIVATE folia::markdown)
